// Parameters
width = 80.0;
depth = 50.36;
height = 27.73;
corner_radius = 6.0;

// Module for a rounded rectangle prism centered at origin in XY, extruded along Z
module rounded_rect_prism(w, d, h, r) {
    // Create a 2D rounded rectangle and extrude it
    // The polygon is defined in XY plane, then extruded along Z
    // We need to place it so the bottom is at Z=0
    translate([0, 0, -h/2]) {
        linear_extrude(height=h) {
            hull() {
                // Four corner circles
                translate([w/2 - r, d/2 - r, 0]) circle(r=r);
                translate([-w/2 + r, d/2 - r, 0]) circle(r=r);
                translate([-w/2 + r, -d/2 + r, 0]) circle(r=r);
                translate([w/2 - r, -d/2 + r, 0]) circle(r=r);
            }
        }
    }
}

// Main design
// Place the solid so its bottom sits on Z=0
translate([0, 0, height/2]) {
    rounded_rect_prism(width, depth, height, corner_radius);
}
